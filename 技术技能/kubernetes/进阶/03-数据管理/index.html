NFS 服务器端配置

步骤 1：安装 NFS 服务器

1# 更新软件包 Ubuntu/Debian 系统
2apt-get update
3
4# 安装 NFS 服务器
5apt-get install -y nfs-kernel-server nfs-common
6
7# 验证安装
8dpkg -l | grep nfs
9systemctl status nfs-kernel-server

步骤 2：创建共享目录

 1# 创建 NFS 共享根目录
 2mkdir -p /data/nfs-share
 3
 4# 创建子目录（可选）
 5mkdir -p /data/nfs-share/pv1
 6mkdir -p /data/nfs-share/pv2
 7mkdir -p /data/nfs-share/backup
 8mkdir -p /data/nfs-share/logs
 9
10# 设置权限（根据需求调整）
11# 方案 1：所有人可读写（开发/测试环境）
12chmod -R 777 /data/nfs-share
13
14# 方案 2：特定用户组（生产环境推荐）
15chown -R nobody:nogroup /data/nfs-share  # Ubuntu/Debian
16# 或
17chown -R nfsnobody:nfsnobody /data/nfs-share  # CentOS/RHEL
18
19chmod -R 755 /data/nfs-share
20
21# 验证
22ls -ld /data/nfs-share

步骤 3：配置 NFS 导出规则

编辑 /etc/exports 文件：

1# 备份原文件
2cp /etc/exports /etc/exports.backup
3
4# 编辑配置
5vi /etc/exports

配置示例

 1# /etc/exports 配置文件
 2# 格式：<共享目录> <客户端>(选项1,选项2,...)
 3
 4# ============ 场景：允许整个子网访问 ============
 5/data/nfs-share 10.0.0.0/24(rw,sync,no_subtree_check,no_root_squash)
 6
 7# ============ 场景：Kubernetes 集群使用（推荐配置） ============
 8/data/nfs-share 10.0.0.0/24(rw,sync,no_subtree_check,no_root_squash,insecure)
 9
10# ============ 危险配置：允许所有主机（不推荐用于生产） ============
11# /data/nfs-share *(rw,sync,no_subtree_check,no_root_squash)

配置选项详解

选项 说明 推荐
rw 读写权限 ✅ 根据需求
ro 只读权限 ✅ 备份目录
sync 同步写入磁盘（数据安全） ✅ 生产必选
async 异步写入（性能优先） ⚠️ 测试环境
no_subtree_check 不检查子目录权限（提高性能） ✅ 推荐
subtree_check 检查子目录权限（安全） ⚠️ 影响性能
no_root_squash root 用户保持权限 ✅ K8s 使用
root_squash root 映射为 nobody（安全） ✅ 一般场景
all_squash 所有用户映射为 nobody ⚠️ 高安全场景
insecure 允许非特权端口连接 ✅ K8s 需要
secure 只允许特权端口 (<1024) ✅ 传统环境
anonuid=1000 匿名用户的 UID ⚠️ 特殊需求
anongid=1000 匿名用户的 GID ⚠️ 特殊需求

步骤 4：应用配置并启动服务

 1# 导出共享目录
 2exportfs -arv
 3
 4# 参数说明：
 5# -a: 导出所有目录
 6# -r: 重新导出所有目录
 7# -v: 显示详细信息
 8
 9# 预期输出：
10# exporting 10.0.0.0/24:/data/nfs-share
11
12# 验证导出列表
13exportfs -v
14
15# 预期输出：
16# /data/nfs-share
17#         10.0.0.0/24(rw,wdelay,no_root_squash,no_subtree_check,sec=sys,rw,secure,no_root_squash,no_all_squash)
18
19# 重启 NFS 服务（确保配置生效）
20systemctl restart nfs-kernel-server  # Ubuntu/Debian
21# 或
22systemctl restart nfs-server         # CentOS/RHEL
23
24# 检查服务状态
25systemctl status nfs-kernel-server   # Ubuntu/Debian
26# 或
27systemctl status nfs-server          # CentOS/RHEL
28
29# 查看 NFS 服务监听端口
30ss -tulnp | grep -E '2049|111'
31# 或
32netstat -tulnp | grep -E '2049|111'
33
34# 预期看到：
35# tcp    0      0 0.0.0.0:2049      0.0.0.0:*       LISTEN      -
36# tcp    0      0 0.0.0.0:111       0.0.0.0:*       LISTEN      -
37# udp    0      0 0.0.0.0:2049      0.0.0.0:*                   -

步骤 5：配置开机自动挂载

方法 1：使用 /etc/fstab（推荐）

1# 备份 fstab
2cp /etc/fstab /etc/fstab.backup
3
4# 编辑 fstab
5vi /etc/fstab
6
7# 添加以下内容（根据实际情况调整）

/etc/fstab 配置示例

 1# /etc/fstab: static file system information.
 2#
 3# <file system>                        <mount point>  <type>  <options>                                           <dump> <pass>
 4
 5# ============ 基础配置（推荐） ============
 610.0.0.100:/data/nfs-share             /mnt/nfs       nfs     defaults,_netdev                                    0      0
 7
 8# ============ NFSv4 配置（性能优化） ============
 910.0.0.100:/data/nfs-share             /mnt/nfs       nfs4    defaults,_netdev,vers=4.2                           0      0
10
11# ============ NFSv3 配置（兼容性） ============
1210.0.0.100:/data/nfs-share             /mnt/nfs       nfs     defaults,_netdev,vers=3                             0      0
13
14# ============ 生产环境配置（高可用） ============
1510.0.0.100:/data/nfs-share             /mnt/nfs       nfs     defaults,_netdev,vers=4,soft,timeo=30,retrans=3     0      0
16
17# ============ 性能优化配置 ============
1810.0.0.100:/data/nfs-share             /mnt/nfs       nfs4    rw,sync,hard,intr,rsize=8192,wsize=8192,timeo=14,_netdev  0  0
19
20# ============ 只读挂载 ============
2110.0.0.100:/data/nfs-share/backup      /mnt/backup    nfs     ro,defaults,_netdev                                 0      0
22
23# ============ 多个挂载点 ============
2410.0.0.100:/data/nfs-share/pv1         /mnt/pv1       nfs4    defaults,_netdev                                    0      0
2510.0.0.100:/data/nfs-share/pv2         /mnt/pv2       nfs4    defaults,_netdev                                    0      0
2610.0.0.100:/data/nfs-share/logs        /var/log/app   nfs4    defaults,_netdev                                    0      0

fstab 选项详解

选项 说明 推荐
defaults 使用默认选项 (rw, suid, dev, exec, auto, nouser, async) ✅ 基础配置
_netdev 等待网络就绪后再挂载（开机必须） ✅ 必须添加
vers=4 使用 NFSv4 协议 ✅ 现代系统
vers=3 使用 NFSv3 协议 ✅ 兼容性
soft 超时后返回错误（不挂起） ✅ 生产推荐
hard 一直重试（挂起直到成功） ⚠️ 高可用
intr 允许中断挂起的操作 ✅ hard 模式必选
timeo=30 超时时间（0.1 秒单位） ✅ 根据网络调整
retrans=3 重试次数 ✅ 默认值
rsize=8192 读取缓冲区大小（字节） ✅ 性能优化
wsize=8192 写入缓冲区大小（字节） ✅ 性能优化
sync 同步写入 ✅ 数据安全
async 异步写入 ⚠️ 性能优先
noatime 不更新访问时间 ✅ 性能优化
nodiratime 不更新目录访问时间 ✅ 性能优化
ro 只读挂载 ✅ 备份目录
rw 读写挂载 ✅ 默认

应用 fstab 配置

 1# 测试 fstab 配置是否正确（不实际挂载）
 2mount -fav
 3
 4# 预期输出：
 5# /mnt/nfs                 : successfully mounted
 6
 7# 实际挂载所有 fstab 中的项
 8mount -av
 9
10# 验证挂载
11df -h | grep nfs
12mount | grep nfs
13
14# 检查挂载选项
15mount | grep /mnt/nfs
16
17# 预期输出：
18# 10.0.0.100:/data/nfs-share on /mnt/nfs type nfs4 (rw,relatime,vers=4.2,...)

方法 2：使用 systemd mount 单元（高级）

创建 systemd mount 文件：

1# 创建 mount 单元文件
2# 文件名格式：<挂载点路径，/ 替换为 ->.mount
3# 例如：/mnt/nfs -> mnt-nfs.mount
4
5vi /etc/systemd/system/mnt-nfs.mount

内容：

 1[Unit]
 2Description=NFS Mount for /mnt/nfs
 3After=network-online.target
 4Wants=network-online.target
 5
 6[Mount]
 7What=10.0.0.100:/data/nfs-share
 8Where=/mnt/nfs
 9Type=nfs4
10Options=defaults,_netdev,vers=4.2
11
12[Install]
13WantedBy=multi-user.target

启用并启动：

 1# 重载 systemd 配置
 2systemctl daemon-reload
 3
 4# 启用开机自动挂载
 5systemctl enable mnt-nfs.mount
 6
 7# 立即挂载
 8systemctl start mnt-nfs.mount
 9
10# 检查状态
11systemctl status mnt-nfs.mount
12
13# 验证
14df -h | grep nfs
