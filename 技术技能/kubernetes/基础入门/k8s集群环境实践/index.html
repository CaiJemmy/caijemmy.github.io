 32# 秘钥文件生成环境
 33create_authkey(){
 34  # 保证历史文件清空
 35  [ -d ${user_dir}/.ssh ] && rm -rf ${user_dir}/.ssh/*
 36  # 构建秘钥文件对
 37  /usr/bin/ssh-keygen -t rsa -P "" -f ${user_dir}/.ssh/id_rsa

 39}
 40# expect自动匹配逻辑
 41expect_autoauth_func(){
 42  # 接收外部参数
 43  command="$@"
 44  expect -c "
 45    spawn ${command}
 46    expect {
 47      \"yes/no\" {send \"yes\r\"; exp_continue}
 48      \"*password*\" {send \"${login_pass}\r\"; exp_continue}
 49      \"*password*\" {send \"${login_pass}\r\"}
 50   }"
 51}
 52# 跨主机传输文件认证
 53sshkey_auth_func(){
 54  # 接收外部的参数
 55  local host_list="$*"
 56  for ip in ${host_list}
 57  do
 58     # /usr/bin/ssh-copy-id -i ${user_dir}/.ssh/id_rsa.pub root@10.0.0.12
 59     cmd="/usr/bin/ssh-copy-id -i ${user_dir}/.ssh/id_rsa.pub"
 60     remote_host="${login_user}@${ip}"
 61     expect_autoauth_func ${cmd} ${remote_host}
 62  done
 63}

 65# 跨主机同步hosts文件
 66scp_hosts_func(){
 67  # 接收外部的参数

 69  for ip in ${host_list}
 70  do
 71     remote_host="${login_user}@${ip}"

 73  done
 74}

 76# 跨主机设定主机名规划
 77set_hostname_func(){
 78  # 接收外部的参数

 81  do
 82     host_name=$(grep ${ip} ${host_file}|awk '{print $NF}')
 83     remote_host="${login_user}@${ip}"
 84     ssh ${remote_host} "hostnamectl set-hostname ${host_name}"
 85  done
 86}

内核调整

简介

1根据kubernetes官方资料的相关信息，我们需要对kubernetes集群是所有主机进行内核参数的调整。

禁用swap

 1	部署集群时，kubeadm默认会预先检查当前主机是否禁用了Swap设备，并在未禁用时强制终止部署过程。因此，在主机内存资源充裕的条件下，需要禁用所有的Swap设备，否则，就需要在后文的kubeadm init及kubeadm join命令执行时额外使用相关的选项忽略检查错误
 2
 3关闭Swap设备，需要分两步完成。
 4首先是关闭当前已启用的所有Swap设备：
 5swapoff -a
 6        
 7而后编辑/etc/fstab配置文件，注释用于挂载Swap设备的所有行。
 8方法一：手工编辑
 9vim /etc/fstab
10# UUID=0a55fdb5-a9d8-4215-80f7-f42f75644f69 none  swap    sw      0       0
11
12方法二：
13sed -i 's/.*swap.*/#&/' /etc/fstab 
14 	替换后位置的&代表前面匹配的整行内容
15注意：
16	只需要注释掉自动挂载SWAP分区项即可，防止机子重启后swap启用
17
18内核(禁用swap)参数
19cat >> /etc/sysctl.d/k8s.conf << EOF
20vm.swappiness=0
21EOF
22sysctl -p /etc/sysctl.d/k8s.conf

1参数解析：
2     --apiserver-advertise-address 要设定为当前集群的master地址，而且必须为ipv4|ipv6地址
3由于kubeadm init命令默认去外网获取镜像，这里我们使用--image-repository来指定使用国内镜像
4	--kubernetes-version选项的版本号用于指定要部署的Kubenretes程序版本，它需要与当前的kubeadm支持的版本保持一致；该参数是必须的
5    --pod-network-cidr选项用于指定分Pod分配使用的网络地址，它通常应该与要部署使用的网络插件（例如flannel、calico等）的默认设定保持一致，10.244.0.0/16是flannel默认使用的网络；
6    --service-cidr用于指定为Service分配使用的网络地址，它由kubernetes管理，默认即为10.96.0.0/12；
7    --ignore-preflight-errors=Swap 如果没有该项，必须保证系统禁用Swap设备的状态。一般最好加上  
8    --image-repository 用于指定我们在安装kubernetes环境的时候，从哪个镜像里面下载相关的docker镜像，如果需要用本地的仓库，那么就用本地的仓库地址即可

1	所以我们有必要对keeaplived进行升级，需要借助于其内部的脚本探测机制实现对后端haproxy进行探测，如果后端haproxy异常就直接把当前的keepalived服务关闭。

制作keepalived的脚本文件

 1[root@localhost ~]# cat /etc/keepalived/check_haproxy.sh
 2#!/bin/bash
 3# 功能: keepalived检测后端haproxy的状态
 4# 版本: v0.1
 5# 作者: 书记
 6# 联系: superopsmsb.com
 7
 8# 检测后端haproxy的状态
 9haproxy_status=$(ps -C haproxy --no-header | wc -l)
10# 如果后端没有haproxy服务，则尝试启动一次haproxy
11if [ $haproxy_status -eq 0 ];then
12        systemctl start haproxy >> /dev/null 2>&1
13        sleep 3
14        # 如果重启haproxy还不成功的话，则关闭keepalived服务
15        if [ $(ps -C haproxy --no-header | wc -l) -eq 0 ]
16        then
17           systemctl stop keepalived
18        fi
19fi
