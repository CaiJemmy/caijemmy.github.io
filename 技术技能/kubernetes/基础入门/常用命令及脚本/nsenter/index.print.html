nsenter 命令

nsenter = Namespace Enter（进入命名空间）

  • Linux 工具，用于进入现有进程的命名空间
  • 主要用于容器调试和故障排查
  • 是 util-linux 包的一部分

基本语法

1nsenter [options] [program [arguments]]

核心选项：

 1-t, --target <PID>    # 指定目标进程的 PID
 2-m, --mount           # 进入 Mount 命名空间
 3-u, --uts             # 进入 UTS 命名空间（hostname）
 4-i, --ipc             # 进入 IPC 命名空间
 5-n, --net             # 进入 Network 命名空间
 6-p, --pid             # 进入 PID 命名空间
 7-U, --user            # 进入 User 命名空间
 8-C, --cgroup          # 进入 Cgroup 命名空间
 9-T, --time            # 进入 Time 命名空间（Linux 5.6+）
10
11-a, --all             # 进入所有命名空间（等同于 -m -u -i -n -p -U）
12-r, --root[=DIR]      # 设置根目录
13-w, --wd[=DIR]        # 设置工作目录

诊断容器网络问题

 1#!/bin/bash
 2# 网络故障诊断
 3
 4CONTAINER=$1
 5PID=$(docker inspect -f '{{.State.Pid}}' $CONTAINER)
 6
 7echo "1. 检查容器 IP"
 8nsenter -t $PID -n ip addr show eth0
 9
10echo ""
11echo "2. 检查默认路由"
12nsenter -t $PID -n ip route | grep default
13
14echo ""
15echo "3. 测试网关连通性"
16GATEWAY=$(nsenter -t $PID -n ip route | grep default | awk '{print $3}')
17nsenter -t $PID -n ping -c 3 $GATEWAY
18
19echo ""
20echo "4. 检查 DNS 配置"
21nsenter -t $PID -m cat /etc/resolv.conf
22
23echo ""
24echo "5. 测试 DNS 解析"
25nsenter -t $PID -n nslookup google.com
26
27echo ""
28echo "6. 测试外网连通性"
29nsenter -t $PID -n ping -c 3 8.8.8.8
30
31echo ""
32echo "7. 检查 NAT 规则"
33nsenter -t $PID -n iptables -t nat -L POSTROUTING -n

抓取容器网络包

 1#!/bin/bash
 2# 容器抓包工具
 3
 4CONTAINER=$1
 5OUTPUT=${2:-/tmp/capture.pcap}
 6
 7if [ -z "$CONTAINER" ]; then
 8    echo "用法: \$0 <容器名> [输出文件]"
 9    exit 1
10fi
11
12PID=$(docker inspect -f '{{.State.Pid}}' $CONTAINER)
13
14echo "开始抓包（按 Ctrl+C 停止）"
15echo "输出文件: $OUTPUT"
16echo ""
17
18# 在容器的网络命名空间中抓包
19nsenter -t $PID -n tcpdump -i eth0 -w $OUTPUT
20
21echo ""
22echo "抓包完成！"
23echo "分析: wireshark $OUTPUT"
24echo "或: tcpdump -r $OUTPUT"

性能分析

 1#!/bin/bash
 2# 容器性能分析
 3
 4CONTAINER=$1
 5PID=$(docker inspect -f '{{.State.Pid}}' $CONTAINER)
 6
 7echo "=== CPU 使用（top 10 进程）==="
 8nsenter -t $PID -p ps aux --sort=-%cpu | head -11
 9
10echo ""
11echo "=== 内存使用（top 10 进程）==="
12nsenter -t $PID -p ps aux --sort=-%mem | head -11
13
14echo ""
15echo "=== 文件描述符使用 ==="
16nsenter -t $PID -m lsof -p 1 | wc -l
17echo "个文件描述符"
18
19echo ""
20echo "=== 网络连接数 ==="
21nsenter -t $PID -n netstat -an | grep ESTABLISHED | wc -l
22echo "个 ESTABLISHED 连接"
23
24echo ""
25echo "=== 磁盘 I/O ==="
26nsenter -t $PID -m iostat -x 1 3
27
28echo ""
29echo "=== 系统调用统计（5秒采样）==="
30nsenter -t $PID -p timeout 5 strace -c -p 1 2>&1 | tail -20
