03、日志记录框架

日志模块

  1#!/usr/bin/env bash
  2#
  3# lib/logging.sh - 生产级日志框架
  4#
  5
  6[[ -n "${_LOGGING_SH:-}" ]] && return
  7readonly _LOGGING_SH=1
  8
  9LOG_DIR="$(dirname "$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)")"
 10LOG_FILE="${LOG_DIR}/logs/shell.log"
 11
 12# 日志配置
 13declare -g LOG_FILE="${LOG_FILE:-}"
 14declare -g LOG_MAX_SIZE="${LOG_MAX_SIZE:-10485760}" # 10MB
 15declare -g LOG_BACKUP_COUNT="${LOG_BACKUP_COUNT:-5}"
 16declare -gi LOG_LEVEL=1  # 0=DEBUG, 1=INFO, 2=WARN, 3=ERROR
 17
 18# 颜色（仅终端）
 19if [[ -t 2 ]]; then
 20    readonly _C_RED=$'\033[31m'
 21    readonly _C_YELLOW=$'\033[33m'
 22    readonly _C_GREEN=$'\033[32m'
 23    readonly _C_BLUE=$'\033[34m'
 24    readonly _C_GRAY=$'\033[90m'
 25    readonly _C_RESET=$'\033[0m'
 26    readonly _C_BOLD=$'\033[1m'
 27else
 28    readonly _C_RED='' _C_YELLOW='' _C_GREEN='' _C_BLUE=''
 29    readonly _C_GRAY='' _C_RESET='' _C_BOLD=''
 30fi
 31
 32# 日志级别设置
 33log_set_level() {
 34    case "${1,,}" in
 35        debug|0) LOG_LEVEL=0 ;;
 36        info|1)  LOG_LEVEL=1 ;;
 37        warn|2)  LOG_LEVEL=2 ;;
 38        error|3) LOG_LEVEL=3 ;;
 39        *) echo "Unknown log level: $1" >&2 ;;
 40    esac
 41}
 42
 43# 设置日志文件
 44log_set_file() {
 45    local file="$1"
 46    local dir
 47    dir="$(dirname "$file")"
 48
 49    if [[ ! -d "$dir" ]]; then
 50        mkdir -p "$dir" || {
 51            echo"Failed to create log directory: $dir" >&2
 52            return 1
 53        }
 54    fi
 55
 56    LOG_FILE="$file"
 57}
 58
 59# 日志轮转
 60_log_rotate() {
 61    [[ -z "$LOG_FILE" ]] && return
 62    [[ ! -f "$LOG_FILE" ]] && return
 63
 64    local size
 65    size=$(stat -f%z "$LOG_FILE" 2>/dev/null || stat -c%s "$LOG_FILE" 2>/dev/null || echo 0)
 66
 67    if (( size > LOG_MAX_SIZE )); then
 68        # 轮转旧文件
 69        for ((i = LOG_BACKUP_COUNT - 1; i >= 1; i--)); do
 70            [[ -f "${LOG_FILE}.$i" ]] && mv "${LOG_FILE}.$i" "${LOG_FILE}.$((i + 1))"
 71        done
 72        mv "$LOG_FILE" "${LOG_FILE}.1"
 73        touch "$LOG_FILE"
 74    fi
 75}
 76
 77# 核心日志函数
 78_log() {
 79    local level=$1 level_name=$2 color=$3
 80    shift 3
 81
 82    (( level < LOG_LEVEL )) && return 0
 83
 84    local timestamp caller_info
 85    timestamp=$(date '+%Y-%m-%d %H:%M:%S.%3N' 2>/dev/null || date '+%Y-%m-%d %H:%M:%S')
 86
 87    # 获取调用位置（跳过内部函数）
 88    local frame=1
 89    while [[ "${FUNCNAME[$frame]:-}" == _log* ]] || [[ "${FUNCNAME[$frame]:-}" == log_* ]]; do
 90        ((frame++))
 91    done
 92    local func="${FUNCNAME[$frame]:-main}"
 93    local line="${BASH_LINENO[$((frame - 1))]:-0}"
 94    caller_info="${func}:${line}"
 95
 96    local message="$*"
 97    local log_line="[${timestamp}] [${level_name}] [${caller_info}] ${message}"
 98
 99    # 输出到终端（带颜色）
100    echo -e "${color}${log_line}${_C_RESET}" >&2
101
102    # 输出到文件（无颜色）
103    if [[ -n "$LOG_FILE" ]]; then
104        echo "$log_line" >> "$LOG_FILE"
105        _log_rotate
106    fi
107}
108
109# 公共日志函数
110log_debug() { _log 0 "DEBUG" "$_C_GRAY"   "$@"; }
111log_info()  { _log 1 "INFO " "$_C_GREEN"  "$@"; }
112log_warn()  { _log 2 "WARN " "$_C_YELLOW" "$@"; }
113log_error() { _log 3 "ERROR" "$_C_RED"    "$@"; }
114
115# 结构化日志（JSON格式）
116log_json() {
117    local level="${1:-info}"
118    local message="${2:-}"
119    shift 2
120
121    local timestamp
122    timestamp=$(date -u '+%Y-%m-%dT%H:%M:%SZ')
123
124    local extra=""
125    while [[ $# -gt 0 ]]; do
126        local key="${1%%=*}"
127        local value="${1#*=}"
128        extra+=", \"${key}\": \"${value}\""
129        shift
130    done
131
132    echo "{\"timestamp\": \"${timestamp}\", \"level\": \"${level}\", \"message\": \"${message}\"${extra}}"
133}
134
135# 执行命令并记录
136log_exec() {
137    local cmd="$*"
138    log_debug "Executing: $cmd"
139
140    local output exit_code
141    output=$("$@" 2>&1) && exit_code=$? || exit_code=$?
142
143    if (( exit_code == 0 )); then
144        log_debug "Command succeeded"
145        [[ -n "$output" ]] && log_debug "Output: $output"
146    else
147        log_error "Command failed with exit code $exit_code"
148        [[ -n "$output" ]] && log_error "Output: $output"
149    fi
150
151    return $exit_code
152}

154# 计时日志
155log_timer_start() {
156    local name="${1:-default}"
157    declare -g "_timer_${name}=$(date +%s%N)"
158}
159
160log_timer_end() {
161    local name="${1:-default}"
162    local start_var="_timer_${name}"
163    local start="${!start_var:-}"
164
165    if [[ -z "$start" ]]; then
166        log_warn "Timer '$name' was not started"
167        return
168    fi
169
170    local end duration_ms
171    end=$(date +%s%N)
172    duration_ms=$(( (end - start) / 1000000 ))
173
174    log_info "Timer '$name': ${duration_ms}ms"
175    unset "$start_var"
176}
