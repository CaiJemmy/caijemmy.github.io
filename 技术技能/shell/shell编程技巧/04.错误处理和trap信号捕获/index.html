04、错误处理和trap信号捕获

trap基础

trap 可以捕获信号并执行指定命令：

注意：

  • ERR trap 不捕获 exit 命令，所以脚本中如果直接使用exit 退出，是不会触发相应trap处理的。
  • 使用 return 替代 exit 可以让 ERR trap 正常工作，此时是脚本整体退出使用的是return的返回值，没有显示exit退出。
  • 函数中的 exit 会直接退出整个脚本，绕过所有 trap。

 2trap 'command' SIGNAL

 4# 常用信号
 5# EXIT  - 脚本退出时（正常或异常）
 6# ERR   - 命令返回非零时（需要set -e）
 7# INT   - Ctrl+C
 8# TERM  - kill命令
 9# HUP   - 终端关闭
10# DEBUG - 每个命令执行前

 67# 错误处理函数
 68error_handler() {
 69    # 保存退出码
 70    local exit_code=$?
 71
 72    # 获取错误信息
 73    local line_no=${1:-$LINENO}
 74    local last_command=${2:-$BASH_COMMAND}
 75    local func_name=${FUNCNAME[1]:-main}
 76    local script_name=$(basename "${BASH_SOURCE[1]:-\$0}")
 77
 78    # 重置trap，避免递归错误
 79    trap - ERR
 80
 81    # 打印错误信息
 82    {
 83        log_info ""
 84        log_info "╔══════════════════════════════════════════════════════════╗"
 85        log_info "║                         ERROR REPORT                     ║"
 86        log_info "╚══════════════════════════════════════════════════════════╝"
 87        log_info ""
 88        log_info "📝 Details:"
 89        log_info "  • Script:    $script_name"
 90        log_info "  • Function:  $func_name"
 91        log_info "  • Line:      $line_no"
 92        log_info "  • Command:   $last_command"
 93        log_info "  • Exit code: $exit_code"
 94        log_info "  • Timestamp: $(date '+%Y-%m-%d %H:%M:%S')"
 95        log_info ""
 96
 97        # 调用栈
 98        if [[ ${#FUNCNAME[@]} -gt 1 ]]; then
 99            log_info "📋 Call stack (most recent first):"
100            for ((i=1; i<${#FUNCNAME[@]}; i++)); do
101                local source_file="${BASH_SOURCE[i]:-unknown}"
102                local line_number="${BASH_LINENO[i-1]:-0}"
103                local function_name="${FUNCNAME[i]:-main}"
104
105                # 美化输出
106                local text=$(printf "  %2d. %s:%d in %s()\n" \
107                    "$((i))" \
108                    "$(basename "$source_file")" \
109                    "$line_number" \
110                    "$function_name")
111                log_info "${text}"
112            done
113            log_info ""
114        fi
115
116        # 相关变量（可选）
117        if [[ -v BASH_ARGV ]]; then
118            log_info "🔧 Function arguments: ${BASH_ARGV[*]}"
119        fi
120
121        log_info ""
122        log_info "💡 Tips:"
123        log_info "  • Check the command syntax"
124        log_info "  • Verify file permissions"
125        log_info "  • Check if required files exist"
126        log_info ""
127        log_info "════════════════════════════════════════════════════════════"
128        log_info ""
129    } >&2
130
131    # 恢复trap
132    trap 'error_handler $LINENO "$BASH_COMMAND"' ERR
133
134    # 返回原始退出码
135    return $exit_code
136}

22# 重试机制
23retry() {
24    local max_attempts="${1:-3}"
25    local delay="${2:-5}"
26    shift 2
27    local cmd=("$@")
28
29    local attempt=1
30    while true; do
31        echo "Attempt $attempt/$max_attempts: ${cmd[*]}" >&2
32
33        if "${cmd[@]}"; then
34            return 0
35        fi
36
37        if (( attempt >= max_attempts )); then
38            echo "All $max_attempts attempts failed" >&2
39            return 1
40        fi
41
42        echo "Failed, retrying in ${delay}s..." >&2
43        sleep "$delay"
44        ((attempt++))
45    done
46}

48# 带超时的命令执行
49run_with_timeout() {
50    local timeout_secs="$1"
51    shift
52
53    # 使用timeout命令（GNU coreutils）
54    if command -v timeout &>/dev/null; then
55        timeout --signal=TERM --kill-after=10 "$timeout_secs" "$@"
56    else
57        # macOS上使用perl实现
58        perl -e '
59            alarm shift @ARGV;
60            exec @ARGV;
61        ' "$timeout_secs" "$@"
62    fi
63}

18        echo"Error: File not found: $file" >&2

28require_root() {

35# 模式4: 互斥锁（防止重复执行）
36acquire_lock() {
37    local lock_file="${1:-/var/lock/$(basename "$0").lock}"
38
39    exec 200>"$lock_file"
40
41    if ! flock -n 200; then
42        echo"Error: Another instance is running" >&2
43        exit 1
44    fi
45
46    # 写入PID
47    echo $$ >&200
48}

50# 模式5: 确认操作
51confirm() {
52    local prompt="${1:-Are you sure?}"
53    local default="${2:-n}"
54
55    local yn_prompt
56    case"$default"in
57        y|Y) yn_prompt="[Y/n]" ;;
58        *)   yn_prompt="[y/N]" ;;
59    esac
60
61    read -rp "$prompt $yn_prompt " response
62    response="${response:-$default}"
63
64    [[ "$response" =~ ^[Yy]$ ]]
65}
