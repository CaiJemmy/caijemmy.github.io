07、单元测试（bats框架）

BATS（Bash Automated Testing System）是一个用 Bash 编写的 TAP（Test Anything Protocol）兼容测试框架，专门用于测试 Bash 脚本和命令行工具。

主要特点：

  • 纯 Bash 实现，无需外部依赖
  • 语法简单，类似 RSpec/Jasmine
  • 支持 TAP 格式输出
  • 可以测试任何命令行程序
  • 支持模拟和桩（stubbing）
  • 并行测试支持

使用 run 命令测试外部命令

当你在 Bats 测试中直接调用一个返回非零退出码的命令时，会发生以下情况：

  1. set -e 生效：Bats 默认启用了 set -e（errexit）
  2. 立即退出：包含 set -e 的脚本在遇到非零退出码的命令时会立即退出当前命令（或函数）
  3. 测试失败：Bats 会将这个视为测试失败，不会继续执行后面的 assert

run 命令是 Bats 的特殊命令，它会：

  1. 在子shell中运行命令
  2. 捕获命令的退出码（存储在 $status 中）
  3. 捕获命令的输出（存储在 $output 中）
  4. 不会因为命令失败而终止测试
 1#!/usr/bin/env bats
 2
 3@test "测试 ls 命令" {
 4    run ls -la
 5    [ "$status" -eq 0 ]
 6    [ "${#lines[@]}" -gt 0 ]
 7    [[ "$output" =~ "total" ]]
 8}
 9
10@test "测试不存在的命令" {
11    run command_that_does_not_exist
12    [ "$status" -ne 0 ]
13    [[ "$output" =~ "not found" || "$output" =~ "command not found" ]]
14}

 4setup_file() {

 53run_coverage() {
 54    local output_dir="coverage"
 55    local test_dir="tests"
 56
 57    rm -rf "$output_dir"
 58    mkdir -p "$output_dir"
 59
 60    # 方法1：使用 find 查找所有测试文件，排除 test_helper 目录
 61    log_info "正在查找测试文件..."
 62
 63    # 获取所有测试文件（排除 test_helper 目录）
 64    local test_files=()
 65    while IFS= read -r -d '' file; do
 66        test_files+=("$file")
 67    done < <(find "$test_dir" -name "*.bats" -type f ! -path "*/test_helper/*" -print0)
 68
 69    # 检查是否找到测试文件
 70    if [[ ${#test_files[@]} -eq 0 ]]; then
 71        log_info "错误: 未找到测试文件"
 72        log_info "在 $test_dir 目录中查找 .bats 文件（排除 test_helper 目录）"
 73        find "$test_dir" -name "*.bats" -type f
 74        return 1
 75    fi
 76
 77    log_info "找到 ${#test_files[@]} 个测试文件:"
 78    log_info '%s\n' "${test_files[@]}"
 79
 80    # 方法2：使用 kcov 运行所有测试文件
 81    log_info "正在使用 kcov 运行测试..."
 82    if command -v kcov >/dev/null 2>&1; then
 83        # 运行 kcov
 84        kcov --include-pattern=".sh" \
 85             --exclude-pattern="tests/test_helper/,$output_dir/,/tmp/" \
 86             "$output_dir" \
 87             bats "${test_files[@]}"
 88
 89        # 检查是否成功
 90        if [[ $? -eq 0 ]] && [[ -f "$output_dir/index.html" ]]; then
 91            log_info ""
 92            log_info "✅ 覆盖率测试完成!"
 93            log_info "📊 覆盖率报告: $(pwd)/$output_dir/index.html"
 94            log_info ""
 95
 96            # 显示简单的覆盖率统计
 97            if [[ -f "$output_dir/coverage.json" ]] && command -v jq >/dev/null 2>&1; then
 98                log_info "📈 覆盖率统计:"
 99                jq -r '
100                    "总行数: " + (.summary.lines_total | tostring) +
101                    ", 覆盖行数: " + (.summary.lines_covered | tostring) +
102                    ", 覆盖率: " + (.summary.percent_covered | tostring) + "%"
103                ' "$output_dir/coverage.json" 2>/dev/null || true
104            fi
105        else
106            log_warn "⚠️  kcov 可能未正确生成报告"
107            log_warn "尝试直接运行测试..."
108
109            # 如果 kcov 失败，直接运行测试
110            bats "${test_files[@]}"
111            return $?
112        fi
113    else
114        log_warn "错误: kcov 未安装"
115        log_warn "请安装 kcov:"
116        log_warn "  Ubuntu/Debian: sudo apt-get install kcov"
117        log_warn "  macOS: brew install kcov"
118        log_warn ""
119        log_warn "将直接运行测试（无覆盖率）..."
120        bats "${test_files[@]}"
121        return $?
122    fi
123}
